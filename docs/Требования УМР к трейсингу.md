# Требования УМР по подключения SDK AEF tracing

## Требования УМР:

- В коде решения подключен сбор трейсов через SDK AEF Tracing
- Начало работы решения начинается с контекстного менеджера aef_input_request (или его аналога, например, aef_agent_start или aef_kafka_produce)
- Финальный результат работы решения записывается в поле aef.response.body спана aef_input_request (или его аналога, например, aef_agent_start)
- При добавлении сообщений в контекст LLM, используются классы из langchain_core.messages: HumanMessage, AIMessage, SystemMessage, FunctionMessage (или аналог, который позволяет идентифицировать типы сообщений в контекстном окне)
- По session_id можно сгруппировать объекты оценки (отдельные запросы или диалоги) для наполнения тестовых датасетов, согласно верифицированному формату на первом этапе валидации
- Подключен трейсинг, для модулей GenAI-решения, чью работу необходимо логировать, согласно верифицированному формату тестового датасета на первом этапе валидаци
- Заполнены все атрибуты в сохраняемых спанах

## На что обратить внимание в примере

Чтобы трейсинг заработал в вашем конкретном случае, нужны следующие действия:
- Инициализация необходимых обработчиков 
- Задание идентификатора сессии `session_id_cvar.set(session_id)`
- Оборачивание каждой операции внутри агента: 
```python
with aef_agent_start(input=request_body) as agent_controller:
      ...
      agent_controller.add_output_result(output={"answer": answer})
```
- Установка и более полные инструкции описаны в https://stash.delta.sbrf.ru/projects/AEF_TEMPLATES/repos/aef-tracing-python/browse?at=v0.4.0 
- Пример генерации тестового сета приведен в файле `create_testset_from_traces.py`; результат - в файле `testset_example.xlsx`

## Пример выполнения требований УМР

```python
import json

from aiohttp import web
from langgraph.graph import StateGraph, MessagesState, START, END
from langchain_core.messages import HumanMessage, AIMessage
from aef_tracing import AEFHandler, AEFTracerProvider, AEFBatchSpanProcessor
from aef_tracing import aef_agent_start, aef_input_request, aef_tool_run, aef_retriever_run, aef_guardrail_run, aef_observation
from aef_tracing.exporters import AEFConsoleSender
from aef_tracing.exporters.json_exporter import AEFJsonSenderExporter
from aef_tracing.span_processors import session_id_cvar

# Инициализация провайдера трассировок
provider = AEFTracerProvider()
# Создание обработчика трассировок
handler = AEFHandler(tracer=provider.get_tracer(__name__))
# Экспорт в консоль
sender = AEFConsoleSender()

# Экспортер с отправкой через JSON
exporter = AEFJsonSenderExporter(senders=[sender])
processor = AEFBatchSpanProcessor(exporter)
provider.add_span_processor(processor)


# --- Функции агента ---

@aef_observation
def preprocess_message(text: str) -> str:
    """Предобработка сообщения (тип спана other)."""
    return text.strip().lower()


def check_guardrail(text: str) -> bool:
    """Проверка guardrail — фильтрация запрещённого контента."""
    forbidden_words = ["запрещено", "опасно"]
    for word in forbidden_words:
        if word in text.lower():
            return False
    return True


def vector_search(query: str) -> list[dict]:
    """Имитация векторного поиска (retriever)."""
    documents = [
        {"page_content": "Python — язык программирования общего назначения.", "metadata": {"source": "wiki"}},
        {"page_content": "LangChain — фреймворк для создания приложений на основе LLM.", "metadata": {"source": "docs"}},
    ]
    return [doc for doc in documents if any(w in doc["page_content"].lower() for w in query.lower().split() if len(w) > 3)]


def external_tool_call(query: str) -> str:
    """Имитация вызова внешнего инструмента."""
    return f"Результат инструмента для запроса: {query}"


# --- Граф LangGraph ---

def mock_llm(state: MessagesState):
    """Имитация вызова LLM через LangGraph (при использовании модели через langchain спан llm собирается автоматически через AEFHandler)."""
    last_message = state["messages"][-1].content
    return {"messages": [AIMessage(content=f"Ответ агента: обработано сообщение '{last_message}'")]}


graph = StateGraph(MessagesState)
graph.add_node("mock_llm", mock_llm)
graph.add_edge(START, "mock_llm")
graph.add_edge("mock_llm", END)
graph = graph.compile()


# --- HTTP-обработчик ---

async def chat_handler(request):
    user_message = request.query.get("message", "Расскажи про Python")
    request_headers = dict(request.headers)
    request_body = {"message": user_message}

    session_id = request.headers.get("x-session-id")
    session_id_cvar.set(session_id)

    # Спан входящего запроса
    with aef_input_request(
        span_name="input_request",
        headers=request_headers,
        body=request_body,
        path="/chat",
        method="GET"
    ) as input_request_controller:

        # Спан запуска агента
        with aef_agent_start(input=request_body) as agent_controller:

            # Спан предобработки (тип other через декоратор)
            processed_message = preprocess_message(user_message)

            # Спан guardrail
            with aef_guardrail_run(input={"text": processed_message}, span_name="content_filter") as guard_span:
                is_allowed = check_guardrail(processed_message)
                guard_span.add_output_result(output={"allowed": is_allowed})

            if not is_allowed:
                agent_controller.add_output_result(output={"answer": "Запрос заблокирован guardrail."})
                input_request_controller.add_response(
                    headers={"Content-Type": "application/json"},
                    body={"answer": "Запрос заблокирован guardrail."},
                    http_code=200,
                )
                return web.Response(text="Запрос заблокирован guardrail.")

            # Спан retriever (векторный поиск)
            with aef_retriever_run(input={"query": processed_message}, span_name="vector_search") as retriever_span:
                documents = vector_search(processed_message)
                retriever_span.add_retriever_result(output=documents)

            # Спан tool (вызов инструмента)
            with aef_tool_run(input={"query": processed_message}, span_name="external_tool") as tool_span:
                tool_result = external_tool_call(processed_message)
                tool_span.add_tool_result(output=tool_result)

            # Вызов LangGraph графа — спаны chain и llm собираются автоматически через AEFHandler
            result = graph.invoke(
                {"messages": [HumanMessage(content=user_message)]},
                config={"callbacks": [handler]}
            )

            answer = result["messages"][-1].content
            agent_controller.add_output_result(output={"answer": answer})

        response_body = {"answer": answer}
        input_request_controller.add_response(
            headers={"Content-Type": "application/json"},
            body=response_body,
            http_code=200,
        )

    return web.Response(text=answer)


app = web.Application()
app.add_routes([web.get('/chat', chat_handler)])

if __name__ == '__main__':
    print("Server starting at http://localhost:8080")
    print('Try: curl "http://localhost:8080/chat?message=Hello"')
    web.run_app(app, port=8080)
```
